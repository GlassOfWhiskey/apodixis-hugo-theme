Homepage templates typically call it as: {{ partial "quickstart.html" site.Params.quickstart }} or, when nothing is configured, it renders nothing. Set an explicitly empty string ("") to render the code with no prompt prefix. code optional code line(s) } } footer { optional call-to-action line under the steps html string raw inline HTML (links, spans, etc.), wrapped in .qs-sub. Use safeHTML on the value so markdown/entities render. } */ -}} {{- $cfg := .data -}} {{- if not (and $cfg (index $cfg "steps")) -}} {{- $cfg = site.Params.quickstart -}} {{- end -}} {{- if not (index $cfg "steps") -}} {{- /* nothing configured → render nothing */ -}} {{- else -}} {{- $title := index $cfg "title" | default "Quickstart" -}} {{- $steps := index $cfg "steps" -}}

{{- end -}}
